    <script>
        // State
        let dataA, dataB;
        let activeMode = 'slider';
        let isFocusMode = false;
        let alignment = { x: 0, y: 0, scale: 1, rotate: 0 };
        let sliderRatio = 0.5;

        // Elements
        const els = {
            fileA: document.getElementById('file-a'),
            fileB: document.getElementById('file-b'),
            visualizer: document.getElementById('visualizer-area'),
            placeholder: document.getElementById('placeholder-text'),
            sliderView: document.getElementById('slider-view'),
            opacityView: document.getElementById('opacity-view'),
            diffView: document.getElementById('diff-view'),
            imgASlider: document.getElementById('img-a-slider'),
            imgBSlider: document.getElementById('img-b-slider'),
            sliderOverlay: document.getElementById('slider-overlay'),
            sliderHandle: document.getElementById('slider-handle'),
            imgAOpacity: document.getElementById('img-a-opacity'),
            imgBOpacity: document.getElementById('img-b-opacity'),
            diffCanvas: document.getElementById('diff-canvas'),
            opacityControl: document.getElementById('opacity-control'),
            opacityRange: document.getElementById('opacity-range'),
            opacityVal: document.getElementById('opacity-val'),
            toggleFocus: document.getElementById('toggle-focus'),
            exportBtn: document.getElementById('export-btn'),
            exportCanvas: document.getElementById('export-canvas'),
            body: document.body
        };

        const controls = {
            x: document.getElementById('range-x'),
            y: document.getElementById('range-y'),
            scale: document.getElementById('range-scale'),
            rotate: document.getElementById('range-rotate')
        };

        // UI Handlers
        els.toggleFocus.onclick = () => {
            isFocusMode = !isFocusMode;
            els.body.classList.toggle('focus-mode', isFocusMode);
            document.getElementById('focus-text').innerText = isFocusMode ? "Menü zeigen" : "Fokus-Modus";
            setTimeout(syncSize, 350);
        };

        els.fileA.onchange = (e) => loadFile(e, 'A');
        els.fileB.onchange = (e) => loadFile(e, 'B');

        function loadFile(e, type) {
            const file = e.target.files[0];
            if (!file) return;
            const reader = new FileReader();
            reader.onload = (ev) => {
                const img = new Image();
                img.onload = () => {
                    if (type === 'A') {
                        dataA = img;
                        els.imgASlider.src = ev.target.result;
                        els.imgAOpacity.src = ev.target.result;
                        document.getElementById('info-a').innerText = `Bild A: ${img.width}x${img.height}px`;
                    } else {
                        dataB = img;
                        els.imgBSlider.src = ev.target.result;
                        els.imgBOpacity.src = ev.target.result;
                        document.getElementById('info-b').innerText = `Bild B: ${img.width}x${img.height}px`;
                    }
                    if (dataA && dataB) {
                        els.placeholder.classList.add('hidden');
                        els.visualizer.classList.remove('hidden');
                        syncSize();
                        updateAlignment();
                    }
                };
                img.src = ev.target.result;
            };
            reader.readAsDataURL(file);
        }

        // KERN-LOGIK: Berechnet das Verhältnis Browser-Anzeige zu Originaldatei
        function getScaleInfo() {
            if (!dataA || !els.imgASlider) return null;
            const container = els.imgASlider;
            const cw = container.clientWidth;
            const ch = container.clientHeight;
            const iw = dataA.width;
            const ih = dataA.height;
            
            // CSS object-contain Logik nachbilden
            const ratio = Math.min(cw / iw, ch / ih);
            const renderedWidth = iw * ratio;
            const renderedHeight = ih * ratio;
            
            return {
                ratio: ratio,
                pixelFactor: iw / renderedWidth, // Faktor Browser-Pixel -> Image-Pixel
                renderedWidth,
                renderedHeight,
                offsetX: (cw - renderedWidth) / 2,
                offsetY: (ch - renderedHeight) / 2
            };
        }

        function updateAlignment() {
            alignment = {
                x: parseInt(controls.x.value),
                y: parseInt(controls.y.value),
                scale: parseInt(controls.scale.value) / 100,
                rotate: parseFloat(controls.rotate.value)
            };

            document.getElementById('val-x').innerText = alignment.x + 'px';
            document.getElementById('val-y').innerText = alignment.y + 'px';
            document.getElementById('val-scale').innerText = Math.round(alignment.scale * 100) + '%';
            document.getElementById('val-rotate').innerText = alignment.rotate + '°';

            const transStr = `translate(${alignment.x}px, ${alignment.y}px) scale(${alignment.scale}) rotate(${alignment.rotate}deg)`;
            els.imgBSlider.style.transform = transStr;
            els.imgBOpacity.style.transform = `translate(calc(-50% + ${alignment.x}px), calc(-50% + ${alignment.y}px)) scale(${alignment.scale}) rotate(${alignment.rotate}deg)`;
            
            if (activeMode === 'diff') renderDiff();
        }

        Object.values(controls).forEach(c => c.oninput = updateAlignment);
        document.getElementById('reset-alignment').onclick = () => {
            controls.x.value = 0; controls.y.value = 0; 
            controls.scale.value = 100; controls.rotate.value = 0;
            updateAlignment();
        };

        // EXPORT: Arbeitet 100% entkoppelt auf dem Offscreen-Canvas
        els.exportBtn.onclick = () => {
            if (!dataA || !dataB) return;
            const info = getScaleInfo();
            const w = dataA.width;
            const h = dataA.height;
            const canvas = els.exportCanvas;
            canvas.width = w; canvas.height = h;
            const ctx = canvas.getContext('2d');

            // Umrechnung des Browser-Offsets in echte Pixel
            const realX = alignment.x * info.pixelFactor;
            const realY = alignment.y * info.pixelFactor;

            if (activeMode === 'diff') {
                renderDiff(true); // Rendert das Diff-Canvas in voller Größe
                ctx.drawImage(els.diffCanvas, 0, 0);
            } else {
                // Basis Zeichnen
                ctx.drawImage(dataA, 0, 0);

                ctx.save();
                // Bild B transformieren
                ctx.translate(w/2 + realX, h/2 + realY);
                ctx.scale(alignment.scale, alignment.scale);
                ctx.rotate(alignment.rotate * Math.PI / 180);

                if (activeMode === 'opacity') {
                    ctx.globalAlpha = els.opacityRange.value / 100;
                    ctx.drawImage(dataB, -dataB.width/2, -dataB.height/2);
                } else if (activeMode === 'slider') {
                    // Temporärer Canvas für Clipping
                    const temp = document.createElement('canvas');
                    temp.width = w; temp.height = h;
                    const tctx = temp.getContext('2d');
                    tctx.save();
                    tctx.translate(w/2 + realX, h/2 + realY);
                    tctx.scale(alignment.scale, alignment.scale);
                    tctx.rotate(alignment.rotate * Math.PI / 180);
                    tctx.drawImage(dataB, -dataB.width/2, -dataB.height/2);
                    tctx.restore();

                    ctx.restore(); ctx.save();
                    // Mapping des Slider-Schnittpunkts
                    const viewWidth = els.sliderView.clientWidth;
                    const sliderX = sliderRatio * viewWidth;
                    const realSliderX = (sliderX - info.offsetX) * info.pixelFactor;

                    ctx.beginPath();
                    ctx.rect(0, 0, realSliderX, h);
                    ctx.clip();
                    ctx.drawImage(temp, 0, 0);
                }
                ctx.restore();
            }

            const link = document.createElement('a');
            link.download = `Analyse_Export_${Date.now()}.png`;
            link.href = canvas.toDataURL('image/png');
            link.click();
        };

        function setMode(m) {
            activeMode = m;
            [els.sliderView, els.opacityView, els.diffView].forEach(v => v.classList.add('hidden'));
            els.opacityControl.classList.toggle('hidden', m !== 'opacity');
            document.querySelectorAll('.mode-btn').forEach(btn => {
                const isActive = btn.id === `mode-${m}`;
                btn.className = `mode-btn w-full text-left px-4 py-3 rounded-xl text-sm font-bold border transition-all ${isActive ? 'bg-blue-600 text-white shadow-md border-blue-600' : 'bg-gray-50 text-gray-700 border-gray-100 hover:bg-gray-100'}`;
            });
            if (m === 'slider') { els.sliderView.classList.remove('hidden'); syncSize(); }
            if (m === 'opacity') els.opacityView.classList.remove('hidden');
            if (m === 'diff') { els.diffView.classList.remove('hidden'); renderDiff(); }
        }

        document.getElementById('mode-slider').onclick = () => setMode('slider');
        document.getElementById('mode-opacity').onclick = () => setMode('opacity');
        document.getElementById('mode-diff').onclick = () => setMode('diff');

        function syncSize() { if (dataA && els.imgASlider) els.imgBSlider.style.width = els.imgASlider.offsetWidth + 'px'; }

        els.sliderView.onmousemove = (e) => {
            const r = els.sliderView.getBoundingClientRect();
            sliderRatio = Math.min(Math.max(((e.clientX - r.left) / r.width), 0), 1);
            els.sliderOverlay.style.width = (sliderRatio * 100) + '%';
            els.sliderHandle.style.left = (sliderRatio * 100) + '%';
        };

        function renderDiff(isExport = false) {
            if (!dataA || !dataB) return;
            const info = getScaleInfo();
            const w = dataA.width, h = dataA.height;
            const realX = alignment.x * info.pixelFactor;
            const realY = alignment.y * info.pixelFactor;

            els.diffCanvas.width = w; els.diffCanvas.height = h;
            const ctx = els.diffCanvas.getContext('2d');
            const cB = document.createElement('canvas'); cB.width = w; cB.height = h;
            const ctxB = cB.getContext('2d');
            
            ctxB.save(); 
            ctxB.translate(w/2 + realX, h/2 + realY); 
            ctxB.scale(alignment.scale, alignment.scale); 
            ctxB.rotate(alignment.rotate * Math.PI / 180); 
            ctxB.drawImage(dataB, -dataB.width/2, -dataB.height/2); 
            ctxB.restore();

            const ctxA = document.createElement('canvas').getContext('2d'); 
            ctxA.canvas.width = w; ctxA.canvas.height = h; 
            ctxA.drawImage(dataA, 0, 0);

            const imgA = ctxA.getImageData(0,0,w,h), imgB = ctxB.getImageData(0,0,w,h), res = ctx.createImageData(w,h);
            for(let i=0; i<imgA.data.length; i+=4) {
                const d = (Math.abs(imgA.data[i]-imgB.data[i]) + Math.abs(imgA.data[i+1]-imgB.data[i+1]) + Math.abs(imgA.data[i+2]-imgB.data[i+2])) / 3;
                if(d > 15) { res.data[i]=255; res.data[i+1]=60; res.data[i+2]=60; res.data[i+3]=255; }
                else { const a = (imgA.data[i]+imgA.data[i+1]+imgA.data[i+2])/3; res.data[i]=res.data[i+1]=res.data[i+2]=a*0.1+35; res.data[i+3]=255; }
            }
            ctx.putImageData(res, 0, 0);
        }

        els.opacityRange.oninput = (e) => { 
            els.imgBOpacity.style.opacity = e.target.value/100; 
            els.opacityVal.innerText = e.target.value+'%'; 
        };

        window.onresize = syncSize;
    </script>
